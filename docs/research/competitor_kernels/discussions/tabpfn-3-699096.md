TabPFN-3 (Prior Labs’ tabular foundation model) has been released. If you work with structured tabular data on Kaggle, it is worth a look: a single-model workflow, strong baselines without long AutoML tuning loops, and on the hosted side options such as thinking mode when you are willing to spend more compute at fit time for better quality.

Official and product

https://priorlabs.ai/tabpfn
https://ux.priorlabs.ai/home

Thinking mode (API / tabpfn-client)

https://docs.priorlabs.ai/capabilities/thinking-mode

Short version: extra inference-time compute during fit. For classification, supported thinking metrics include accuracy, log_loss, and roc_auc; for regression, rmse and mae. Quotas and limits are described on that page.

Technical deep dive

https://storage.googleapis.com/prior-labs-tabpfn-public/reports/TabPFN_3_model_report.pdf

Open-source inference package

https://github.com/PriorLabs/TabPFN/releases/tag/v8.0.0

Weights and model card

https://huggingface.co/Prior-Labs/tabpfn_3

# pip install -q tabpfn-client
import numpy as np
import pandas as pd
from tabpfn_client import TabPFNClassifier, set_access_token

set_access_token("your token")

train = pd.read_csv("train.csv")
test = pd.read_csv("test.csv")
sample_submission = pd.read_csv("sample_submission.csv")

target = "PitNextLap"
drop_cols = ["id"]
feature_cols = [c for c in train.columns if c not in drop_cols + [target]]

X, y = train[feature_cols], train[target]
X_test = test[feature_cols]

# for thinking mode, you can use max 200k train data
model = TabPFNClassifier(
    random_state=42,
    # thinking_mode=True,
    # thinking_effort="high",
    # thinking_metric="roc_auc",
    # thinking_timeout_s=300,

)

model.fit(X, y)

BATCH = 50_000
test_pred = np.concatenate(
    [model.predict(X_test.iloc[s : s + BATCH]) for s in range(0, len(X_test), BATCH)]
)

sub = sample_submission.copy()
sub[target] = test_pred
sub.to_csv("submission_tabpfn_client.csv", index=False)

```

def BatchedPredicting(Model, batch_size=50000):
    class BatchedPredictingModel(Model):
        def predict_(self, X, method):
            n = int(np.ceil(len(X)/batch_size))
            batches = np.array_split(np.arange(len(X)), n)
            p = getattr(super(), method)
            return np.concatenate(
                [p(X.iloc[I]) for I in batches],
                axis=0
            )
        def predict(self, X): return self.predict_(X, 'predict')
        def predict_proba(self, X): return self.predict_(X, 'predict_proba')

    return BatchedPredictingModel

from sklearn.model_selection import cross_val_score, StratifiedKFold
from sklearn.metrics import make_scorer, roc_auc_score
from tabpfn import TabPFNClassifier, TabPFNRegressor

X = pd.read_csv('train.csv', index_col='id')
y = X.pop('PitNextLap')
kfold = StratifiedKFold(5, shuffle=True, random_state=0)

def get_model(ckpt):
    Model = TabPFNClassifier if 'classifier' in ckpt else TabPFNRegressor
    return BatchedPredicting(Model)(
        ignore_pretraining_limits=True,
        model_path=F'/kaggle/working/{ckpt}.ckpt',
        device='cuda',
        fit_mode='fit_with_cache'
    )

models = {
    'regressor (default)': get_model(
        'tabpfn-v3-regressor-v3_default'
    ),
    'classifier (default)': get_model(
        'tabpfn-v3-classifier-v3_default'
    ),
    'classifier (binary)': get_model(
        'tabpfn-v3-classifier-v3_20260417_binary'
    ),
}

for m, model in models.items():
    scoring = 'roc_auc' if 'classifier' in m else make_scorer(
        roc_auc_score, response_method='predict'
    )
    scores = cross_val_score(
        model, X, y, cv=kfold,
        scoring=scoring, n_jobs=1,
        error_score='raise'
    )
    print(F'{m}: {scores.mean():.5f} ± {scores.std():.5f}')
```
